100 def reshape(self, data, x, y, transpose = False):
101 """Converts a System.Double[,] to a 2D list for plotting or post processing
102
103 Parameters
104 ----------
105 data : System.Double[,] data directly from ZOS-API
106 x : x width of new 2D list [use var.GetLength(0) for dimension]
107 y : y width of new 2D list [use var.GetLength(1) for dimension]
108 transpose : transposes data; needed for some multi-dimensional line series data
109
110 Returns
111 -------
112 res : 2D list; can be directly used with Matplotlib or converted to
113 a numpy array using numpy.asarray(res)
114 """
115 if type(data) is not list:
116 data = list(data)
117 var_lst = [y] * x;
118 it = iter(data)
119 res = [list(islice(it, i)) for i in var_lst]
120 if transpose:
121 return self.transpose(res);
122 return res

132
133 Returns

135 res : transposed 2D list
136 """
137 if type(data) is not list:
138 data = list(data)
