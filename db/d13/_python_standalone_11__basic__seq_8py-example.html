99 def reshape(self, data, x, y, transpose = False):
100 """Converts a System.Double[,] to a 2D list for plotting or post processing
101
102 Parameters
103 ----------
104 data : System.Double[,] data directly from ZOS-API
105 x : x width of new 2D list [use var.GetLength(0) for dimension]
106 y : y width of new 2D list [use var.GetLength(1) for dimension]
107 transpose : transposes data; needed for some multi-dimensional line series data
108
109 Returns
110 -------
111 res : 2D list; can be directly used with Matplotlib or converted to
112 a numpy array using numpy.asarray(res)
113 """
114 if type(data) is not list:
115 data = list(data)
116 var_lst = [y] * x;
117 it = iter(data)
118 res = [list(islice(it, i)) for i in var_lst]
119 if transpose:
120 return self.transpose(res);
121 return res
122
123 def transpose(self, data):
124 """Transposes a 2D list (Python3.x or greater).
125
126 Useful for converting mutli-dimensional line series (i.e. FFT PSF)
127
128 Parameters
129 ----------
130 data : Python native list (if using System.Data[,] object reshape first)
131
132 Returns
133 -------
134 res : transposed 2D list
135 """
136 if type(data) is not list:
137 data = list(data)
138 return list(map(list, zip(*data)))
