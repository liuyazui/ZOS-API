C#

// change MCE to NPAR, modifies the number of Layout Rays for a Source
for (int a = 1; a <= 4; a++)
{
TheMCE.GetOperandAt(a).ChangeType(ZOSAPI.Editors.MCE.MultiConfigOperandType.NPAR);
}

TheMCE.GetOperandAt(1).Param3 = 1;
