101 def reshape(self, data, x, y, transpose = False):
102 """Converts a System.Double[,] to a 2D list for plotting or post processing
103
104 Parameters
105 ----------
106 data : System.Double[,] data directly from ZOS-API
107 x : x width of new 2D list [use var.GetLength(0) for dimension]
108 y : y width of new 2D list [use var.GetLength(1) for dimension]
109 transpose : transposes data; needed for some multi-dimensional line series data
110
111 Returns
112 -------
113 res : 2D list; can be directly used with Matplotlib or converted to
114 a numpy array using numpy.asarray(res)
115 """
116 if type(data) is not list:
117 data = list(data)
118 var_lst = [y] * x;
119 it = iter(data)
120 res = [list(islice(it, i)) for i in var_lst]
121 if transpose:
122 return self.transpose(res);
123 return res
124
125 def transpose(self, data):
126 """Transposes a 2D list (Python3.x or greater).
127
128 Useful for converting mutli-dimensional line series (i.e. FFT PSF)
129
130 Parameters
131 ----------
132 data : Python native list (if using System.Data[,] object reshape first)
133
134 Returns
135 -------
136 res : transposed 2D list
137 """
138 if type(data) is not list:
139 data = list(data)
140 return list(map(list, zip(*data)))
