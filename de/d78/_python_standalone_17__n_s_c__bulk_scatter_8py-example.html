97 def reshape(self, data, x, y, transpose = False):
98 """Converts a System.Double[,] to a 2D list for plotting or post processing
99
100 Parameters
101 ----------
102 data : System.Double[,] data directly from ZOS-API
103 x : x width of new 2D list [use var.GetLength(0) for dimension]
104 y : y width of new 2D list [use var.GetLength(1) for dimension]
105 transpose : transposes data; needed for some multi-dimensional line series data
106
107 Returns
108 -------
109 res : 2D list; can be directly used with Matplotlib or converted to
110 a numpy array using numpy.asarray(res)
111 """
112 if type(data) is not list:
113 data = list(data)
114 var_lst = [y] * x;
115 it = iter(data)
116 res = [list(islice(it, i)) for i in var_lst]
117 if transpose:
118 return self.transpose(res);
119 return res

121 def transpose(self, data):
122 """Transposes a 2D list (Python3.x or greater).
123
124 Useful for converting mutli-dimensional line series (i.e. FFT PSF)
125
126 Parameters
127 ----------
128 data : Python native list (if using System.Data[,] object reshape first)
129
130 Returns
131 -------
132 res : transposed 2D list
133 """
134 if type(data) is not list:
135 data = list(data)
136 return list(map(list, zip(*data)))
