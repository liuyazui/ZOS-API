Object_2.Material = "N-BK7";

RectVolume2_data.X2HalfWidth = 12;

RectVolume2_volphysdata.ModelSettings._S_AngleScattering.Angle = 30;

RectVolume2_Drawdata.Opacity = ZOSAPI.Common.ZemaxOpacity.P50;
