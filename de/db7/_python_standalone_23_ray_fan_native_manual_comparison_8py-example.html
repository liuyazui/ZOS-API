103 def reshape(self, data, x, y, transpose = False):
104 """Converts a System.Double[,] to a 2D list for plotting or post processing
105
106 Parameters
107 ----------
108 data : System.Double[,] data directly from ZOS-API
109 x : x width of new 2D list [use var.GetLength(0) for dimension]
110 y : y width of new 2D list [use var.GetLength(1) for dimension]
111 transpose : transposes data; needed for some multi-dimensional line series data
112
113 Returns
114 -------
115 res : 2D list; can be directly used with Matplotlib or converted to
116 a numpy array using numpy.asarray(res)
117 """
118 if type(data) is not list:
119 data = list(data)
120 var_lst = [y] * x;
121 it = iter(data)
122 res = [list(islice(it, i)) for i in var_lst]
123 if transpose:
124 return self.transpose(res);
125 return res

127 def transpose(self, data):
128 """Transposes a 2D list (Python3.x or greater).
129
130 Useful for converting mutli-dimensional line series (i.e. FFT PSF)
131
132 Parameters
133 ----------
134 data : Python native list (if using System.Data[,] object reshape first)
135
136 Returns
137 -------
138 res : transposed 2D list
139 """
140 if type(data) is not list:
141 data = list(data)
142 return list(map(list, zip(*data)))
